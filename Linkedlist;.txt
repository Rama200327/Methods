using System;
class program
{
    static void main()
    {
        //Even Numbers Filter
        List<int> numbers = new List<int> { 1, 2, 3, 4, 5, 6 };
        List<int> evens = numbers.Where(n => n % 2 == 0).ToList();

        //Remove Duplicates
        List<int> duplicates = new List<int> { 1, 2, 2, 3, 4, 4, 5 };
        List<int> unique = duplicates.Distinct().ToList();

        //Find Maximum Number
        int maxNum = numbers.Max();

        //Reverse List
        numbers.Reverse(); // Built-in method

        //Count Numbers Greater Than 50
        List<int> bigNumbers = new List<int> { 10, 60, 30, 80, 100 };
        int count = bigNumbers.Count(n => n > 50);


        //Student Scores (> 80)
        Dictionary<string, int> scores = new Dictionary<string, int> { { "Ali", 85 }, { "Sara", 70 }, { "Omar", 90 } };
        var highScores = scores.Where(s => s.Value > 80);

        //Search by Key
        string searchName = "Ali";
        if (scores.ContainsKey(searchName))
            Console.WriteLine($"{searchName}: {scores[searchName]}");
        else
            Console.WriteLine("Not found");

        //Sum of Values
        int totalScores = scores.Values.Sum();

        //Remove Low Scores (< 50)
        var keysToRemove = scores.Where(s => s.Value < 50).Select(s => s.Key).ToList();
        foreach (var key in keysToRemove) scores.Remove(key);

        //Highest Score Student
        var topStudent = scores.OrderByDescending(s => s.Value).First();

        LinkedList<string> nav = new LinkedList<string>();

        //Add Elements
        nav.AddFirst("Home");
        nav.AddLast("Contact");

        //Remove Middle
        var middleNode = nav.Find("Home"); // Example finding
        if (middleNode != null) nav.Remove(middleNode);

        //Add Before/After
        var node = nav.Find("Home");
        nav.AddAfter(node, "About");

        //Traverse
        foreach (var item in nav) Console.WriteLine(item);

        //First and Last
        Console.WriteLine($"First: {nav.First.Value}, Last: {nav.Last.Value}");

        int[] data = { 10, 20, 150, 35, 200, 15 };

        //Odd Numbers
        var odds = data.Where(n => n % 2 != 0);

        //Multiply All Numbers by 2
        var doubled = data.Select(n => n * 2);

        //Sort Descending
        var sorted = data.OrderByDescending(n => n);

        //First Number Greater Than 100
        int firstHigh = data.FirstOrDefault(n => n > 100);

        //Average
        double avg = data.Average();
    }
}
